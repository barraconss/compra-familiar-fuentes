 // Privacidad y memoria: convertir las fotos a un archivo WebP nuevo sin metadatos EXIF.
 // No se reduce por debajo de 2200px la mayor dimension si la foto ya es menor.
 // Solo formatos raster comprobables; otros tipos de imagen se rechazan en vez de guardar EXIF.
 async function prepareReceiptImage(file){
   if(isPDF(file))return {file,compressed:false};
   const mime=String(file.type||'').toLowerCase();
   if(!['image/jpeg','image/png','image/webp'].includes(mime))throw Error('Imagen no compatible para eliminar metadatos. Adjunta JPEG, PNG o WebP, o utiliza PDF.');
   const bitmap=await (async()=>{
     if(typeof createImageBitmap==='function')return await createImageBitmap(file,{imageOrientation:'from-image'});
     return await new Promise((resolve,reject)=>{const url=URL.createObjectURL(file),im=new Image();im.onload=()=>{URL.revokeObjectURL(url);resolve(im);};im.onerror=()=>{URL.revokeObjectURL(url);reject(Error('No se pudo abrir la imagen'));};im.src=url;});
   })();
   try{
     const width=bitmap.width||bitmap.naturalWidth,height=bitmap.height||bitmap.naturalHeight;
     if(!width||!height||width*height>60000000)throw Error('La imagen es demasiado grande para procesarla de forma segura.');
     const scale=Math.min(1,2200/Math.max(width,height));
     const c=document.createElement('canvas');c.width=Math.max(1,Math.round(width*scale));c.height=Math.max(1,Math.round(height*scale));
     const ctx=c.getContext('2d',{alpha:false});if(!ctx)throw Error('No se puede procesar la imagen en este dispositivo.');
     ctx.fillStyle='#fff';ctx.fillRect(0,0,c.width,c.height);ctx.drawImage(bitmap,0,0,c.width,c.height);
     const blob=await new Promise((resolve,reject)=>c.toBlob(x=>x?resolve(x):reject(Error('No se ha podido generar la imagen WebP')),'image/webp',0.92));
     if(blob.type!=='image/webp')throw Error('El navegador no admite guardar imágenes WebP.');
     if(blob.size>15*1024*1024)throw Error('La imagen resultante supera 15 MB.');
     const base=(file.name||'ticket').replace(/\.[^.]+$/,'').slice(0,100);
     const newFile=new File([blob],base+'.webp',{type:'image/webp',lastModified:Date.now()});
     return {file:newFile,compressed:true,fromBytes:file.size,toBytes:newFile.size};
   }finally{if(typeof bitmap.close==='function')bitmap.close();}
 }

   try{if(navigator.storage?.persist){return await navigator.storage.persist();}}catch(_){/* no soportado o denegado */}

 function preview(file){clearPreview();if(!file)return;const pdf=isPDF(file),picture=file.type.startsWith('image/');if(!picture&&!pdf){message('Selecciona una imagen o un PDF.');return;}if(file.size>15*1024*1024){message('El archivo supera 15 MB. Comercio seleccionado: '+(($('receipt-store').value==='OTRO'?$('receipt-other-shop').value.trim()||'otra tienda (pendiente de nombre)':$('receipt-store').value||'sin identificar'))+'.'));const toolbar=make('div');toolbar.className='toolbar';toolbar.append(button('Guardar en este dispositivo',save,true),button('Cancelar',clearPreview));node.append(toolbar);node.hidden=false;message('Archivo seleccionado; todavía no está guardado.');}
 async function save(){
  if(!pending)return;
  if($('receipt-store').value==='OTRO'&&!$('receipt-other-shop').value.trim()){message('Escribe el nombre de la tienda antes de guardar.');return;}
  const original=pending;
  message('Preparando justificante y eliminando metadatos de imágenes…');
  try{
    const prepared=await prepareReceiptImage(original),file=prepared.file;
    const rec={id:(window.crypto?.randomUUID?.()||String(Date.now())+'-'+Math.random()),name:file.name||'Justificante de compra',mime:file.type,date:Date.now(),shop:($('receipt-store').value==='OTRO'?$('receipt-other-shop').value.trim():$('receipt-store').value)||null,blob:file,processed:prepared.compressed?'webp-no-exif':'pdf-original',originalSize:original.size};
    await transaction('readwrite','put',rec);
    const persistent=await requestPersistentStorage();
    clearPreview();$('receipt-other-shop').value='';
    const saving=prepared.compressed?'Imagen procesada: '+Math.round(prepared.fromBytes/1024)+' KB → '+Math.round(prepared.toBytes/1024)+' KB. ':'PDF original conservado. ';
    message(saving+'Justificante guardado en este dispositivo. '+(persistent?'Almacenamiento persistente concedido.':'No hay garantía de almacenamiento persistente: exporta una copia ZIP.'));
    await render();
  }catch(err){message('No se ha guardado el justificante. '+(err?.message||'Comprueba espacio y permisos.'));}
 }
