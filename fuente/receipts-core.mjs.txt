/* Justificantes (v0.13, candidata, NO desplegada). Logica pura y sin red: tipo REAL por firma, limites y huella SHA-256.
   No hace OCR ni interpreta importes. La subida a Storage y la UI se conectan aparte (requieren migracion 008 autorizada). */
export const MAX_BYTES=5*1024*1024,MAX_PER_PURCHASE=10;
export const TYPES={'image/jpeg':'jpg','image/png':'png','image/webp':'webp','application/pdf':'pdf'};
const eq=(b,o,s)=>s.every((v,i)=>b[o+i]===v);
/* Detecta el tipo por los primeros bytes; ignora el nombre y el tipo declarado. */
export function sniffType(bytes){
 const b=bytes instanceof Uint8Array?bytes:new Uint8Array(bytes);
 if(b.length>=3&&eq(b,0,[0xFF,0xD8,0xFF]))return 'image/jpeg';
 if(b.length>=8&&eq(b,0,[0x89,0x50,0x4E,0x47,0x0D,0x0A,0x1A,0x0A]))return 'image/png';
 if(b.length>=12&&eq(b,0,[0x52,0x49,0x46,0x46])&&eq(b,8,[0x57,0x45,0x42,0x50]))return 'image/webp';
 if(b.length>=5&&eq(b,0,[0x25,0x50,0x44,0x46,0x2D]))return 'application/pdf';
 return null;
}
export async function sha256Hex(bytes){const d=await crypto.subtle.digest('SHA-256',bytes);return [...new Uint8Array(d)].map(x=>x.toString(16).padStart(2,'0')).join('');}
export function cleanName(name,mime){const base=String(name||'justificante').replace(/[\\/:*?"<>|\u0000-\u001f]+/g,'_').replace(/\.[A-Za-z0-9]{1,5}$/,'').trim().slice(0,100)||'justificante';return base+'.'+TYPES[mime];}
/* Valida un archivo antes de reservar/subir. Devuelve {ok,mime,size,name,sha256} o {ok:false,error}. */
export async function validateReceipt(file,{existing=0}={}){
 if(!file||typeof file.arrayBuffer!=='function')return {ok:false,error:'No hay archivo.'};
 if(existing>=MAX_PER_PURCHASE)return {ok:false,error:'Esta compra ya tiene el maximo de '+MAX_PER_PURCHASE+' justificantes.'};
 if(file.size<1)return {ok:false,error:'El archivo esta vacio.'};
 if(file.size>MAX_BYTES)return {ok:false,error:'El archivo supera 5 MB. Reduce la foto o usa un PDF mas ligero.'};
 const bytes=new Uint8Array(await file.arrayBuffer()),mime=sniffType(bytes);
 if(!mime)return {ok:false,error:'Formato no admitido. Usa JPG, PNG, WebP o PDF.'};
 if(file.type&&Object.hasOwn(TYPES,file.type)&&file.type!==mime)return {ok:false,error:'El contenido no coincide con el tipo declarado del archivo.'};
 return {ok:true,mime,size:bytes.length,name:cleanName(file.name,mime),sha256:await sha256Hex(bytes),bytes};
}
/* Tras descargar: la huella debe coincidir con la guardada en la base. */
export async function verifyDownload(bytes,expectedSha){return (await sha256Hex(bytes))===expectedSha;}
/* Copia de seguridad: lista de comprobacion archivo a archivo (metadatos exportados vs. archivos presentes). */
export async function verifyBackupFiles(receipts,files){const out={ok:0,missing:[],corrupt:[]};for(const r of receipts.filter(x=>x.status==='ready')){const f=files.get(r.id);if(!f){out.missing.push(r.id);continue;}if(!(await verifyDownload(f,r.sha256)))out.corrupt.push(r.id);else out.ok++;}return out;}
