 // la suma de lineas solo SUMA apoyo a un candidato que ya existe (nunca crea uno ni cuenta como conflicto si no coincide)
 if(items.length>=2||(items.length===1&&items[0].nameKnown&&items[0].conf>=85&&!items[0].doubts.length))for(const [v,s] of sup)if(recon(v).exact)s.add('suma de líneas');
 const cand=[...sup].map(([v,s])=>({cents:v,sources:[...s],labels:[...new Set(res.totals.filter(t=>t.cents===v&&t.label).map(t=>t.label))]})).sort((a,b)=>b.sources.length-a.sources.length);
 let total=null,tconf='NINGUNA',conflict=false;
 if(cand.length){const best=cand[0],second=cand[1];
  if(second&&second.sources.length>=best.sources.length){conflict=true;tconf='DUDOSA';addDoubt('total','Hay importes distintos candidatos a total ('+cand.map(c=>fmt(c.cents)+' €').join(' / ')+'): no se elige ninguno por ti.');}
  else{total=best.cents;if(second){conflict=true;addDoubt('total','Hay otro importe distinto del total ('+fmt(second.cents)+' €) en el ticket: confirma el total.');}
   const strong=best.sources.length>=2&&!second;tconf=strong?'VERIFICADA':'PROBABLE';
   // un total con UNA sola fuente y leido con poca seguridad (o con caracteres corregidos) no se propone: queda solo como candidato
   const lt=res.totals.find(t=>t.cents===total);if(lt&&(lt.fixed||lt.numConf<NUM_MIN_CONF)){addDoubt('total','El importe del total se leyó con poca seguridad o se corrigió una confusión de caracteres.');if(best.sources.length<2)tconf='DUDOSA';}}}
 if(tconf==='DUDOSA')total=null;
 res.total={cents:total,conf:total===null?'NINGUNA':tconf,sources:total===null?[]:cand.find(c=>c.cents===total).sources,candidates:total===null?cand:cand.filter(c=>c.cents===total||!(c.sources.length===1&&c.sources[0]==='desglose de IVA'))};/* v0.18: con total elegido, una cifra que solo procede del desglose de IVA no se ofrece como alternativa (el aviso de «otro importe» se mantiene) */
 if(total===null&&!cand.length)addDoubt('total','No se encuentra un importe total fiable en el ticket.');
 let r=total===null?{exact:false,variant:'',net:items.length?isum-dsum:null}:recon(total);
 if(total!==null&&!r.exact&&!res.discounts.length){const alt=altLines(L);if(alt&&alt.length&&alt.reduce((a,i)=>a+i.subtotalCents,0)===total){items.length=0;items.push(...alt);res.items=items;isum=items.reduce((a,i)=>a+i.subtotalCents,0);r=recon(total);}}
 res.hints=[];try{res.hints=hintsFrom(L);}catch{}
 res.reconciliation={items:isum,discounts:dsum,net:r.net,total,diff:total===null||r.net===null?null:total-r.net,exact:r.exact,variant:r.variant,tendered:tendered??null,change:res.change.length?change:null,tax:res.tax.length?res.tax.reduce((a,b)=>a+b,0):null,discountsIgnored:!!r.discountsIgnored};
 if(r.discountsIgnored)for(const d of res.discounts)d.ignored=true;
 if(total!==null&&tendered!==undefined&&tendered!==total&&(!res.change.length||change===0))addDoubt('tendered','El importe ENTREGADO ('+fmt(tendered)+' €) no coincide con el total y no hay cambio: probable error de lectura; no se usa.');
 if(total!==null&&!r.exact&&items.length)addDoubt('lines','Las líneas leídas ('+fmt(isum-dsum)+' €) no suman el total ('+fmt(total)+' €): el desglose no es fiable.');
 // ---- nivel
 let level;
 if(total===null)level='insufficient';
 else if(items.length&&r.exact){const bad=items.some(i=>!i.nameKnown||i.doubts.length)||res.discounts.some(d=>d.doubts.length||d.fixed||d.numConf<NUM_MIN_CONF)||!!r.discountsIgnored||res.date.conf==='DUDOSA';level=(!bad&&tconf==='VERIFICADA')?'complete':'partial';}
 else level='total_only';
 res.level=level;res.levelLabel=LEVELS[level];
 return res;}

/* Varias lecturas del MISMO ticket (distintas segmentaciones de pagina): se elige la mejor (total mas fiable, luego mas completa). Si dos lecturas dan
   totales DISTINTOS y el elegido no esta VERIFICADO, el total se bloquea (no se elige ninguno por la persona). Nunca crea datos nuevos. */
const CONF_RANK={VERIFICADA:3,PROBABLE:2,DUDOSA:1,NINGUNA:0},LEVEL_RANK={complete:3,partial:2,total_only:1,insufficient:0};
export function bestOf(sets,opts={}){const rs=sets.map(l=>({lines:l,res:parseReceipt(l,opts)}));let bi=0;
 for(let i=1;i<rs.length;i++){const a=rs[bi].res,b=rs[i].res,ka=CONF_RANK[a.total.conf]*10+LEVEL_RANK[a.level],kb=CONF_RANK[b.total.conf]*10+LEVEL_RANK[b.level];if(kb>ka)bi=i;}
 const best=rs[bi],others=rs.filter((_,i)=>i!==bi).map(x=>x.res.total.cents).filter(c=>c!==null&&c!==best.res.total.cents);
 if(others.length){const r=best.res;r.doubts.push({field:'total',reason:'Otra lectura del mismo ticket da un total distinto ('+[...new Set(others)].map(fmt).join(' / ')+' €): confirma el total con el papel.'});
  if(r.total.conf!=='VERIFICADA'&&r.total.cents!==null){r.total.candidates.forEach(()=>{});r.total={...r.total,cents:null,conf:'NINGUNA',sources:[]};r.reconciliation={...r.reconciliation,total:null,diff:null,exact:false};r.level='insufficient';r.levelLabel=LEVELS.insufficient;}}
 best.index=bi;best.passes=rs.length;return best;}
/* Resultado -> datos para el formulario de compra existente (comercio, fecha, total, lineas). No guarda nada. */
export function toDraft(res){
 const d={shop:res.shop.value,date:res.date.value,total:res.total.cents!==null?fmt(res.total.cents):'',lines:[],note:'',level:res.level};
 if(res.level==='complete'||res.level==='partial'){for(const i of res.items)d.lines.push({name:i.name,qty:String(i.qty).replace('.',','),price:fmt(i.unitCents),adjustment:false});
  if(!res.reconciliation.discountsIgnored)for(const x of res.discounts)d.lines.push({name:x.name,qty:'1',price:fmt(x.cents),adjustment:true});
  d.note=res.level==='partial'?'Leído de un ticket (OCR) y confirmado por una persona. Algunos artículos sin identificar.':'Leído de un ticket (OCR) y confirmado por una persona.';}
 else if(res.level==='total_only'){d.note='Total leído de un ticket (OCR) y confirmado por una persona; desglose no validado.';}
 return d;}
/* Marcas de duda por campo para resaltar en pantalla. */
export function doubtFlags(res){const f={total:res.total.conf!=='VERIFICADA',shop:res.shop.conf!=='PROBABLE',date:res.date.conf!=='PROBABLE',lines:res.items.map(i=>({name:!i.nameKnown,price:i.doubts.some(d=>!/^nombre ilegible/.test(d))})),reasons:res.doubts.map(x=>x.reason)};return f;}
