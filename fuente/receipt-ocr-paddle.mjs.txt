/* v0.18 CANDIDATA (NO desplegada, apagada por defecto): lector AVANZADO de tickets con PaddleOCR.js (PP-OCRv6 tiny) EN EL DISPOSITIVO.
   Todo se sirve desde este mismo origen (./vendor/paddle/, sin Baidu ni CDN) y NADA se descarga hasta que la persona pulsa «Descargar»:
   los ficheros se guardan en Cache Storage («cf-ocr-paddle-v1», verificados por tamano y SHA-256) y el service worker los sirve desde ahi (funciona sin conexion).
   Carga bajo demanda (import dinamico + worker), un solo hilo, imagen limitada a 1280 px por el lado largo, y el motor se LIBERA siempre al terminar, cancelar o expirar.
   Devuelve el mismo formato que receipt-ocr-engine.mjs (lines/passes/quality/stats) para que el analizador y las protecciones sean las mismas. */
import {sniffType} from './receipts-core.mjs';
import {rowsFromBoxes} from './receipt-ocr-core.mjs';
export const PADDLE_CACHE='cf-ocr-paddle-v1';
export const BASE=new URL('./vendor/paddle/',import.meta.url).href;
export const PADDLE_LIMITS={maxBytes:12*1024*1024,sideLen:1280,timeoutMs:120000,minDeviceMemoryGB:4,disposeMs:4000};
const TYPES={mjs:'text/javascript',js:'text/javascript',wasm:'application/wasm',tar:'application/x-tar',json:'application/json'};
const ctype=p=>TYPES[p.split('.').pop()]||'application/octet-stream';
const mem=()=>{try{return performance.memory?Math.round(performance.memory.usedJSHeapSize/1048576):null;}catch{return null;}};
/* ¿Puede este dispositivo usar el lector avanzado? (no mide memoria real: deviceMemory es una cota gruesa; si falla, el router cae al lector estandar) */
export function paddleSupport(){const why=[];if(typeof Worker==='undefined')why.push('sin Web Workers');if(typeof WebAssembly==='undefined')why.push('sin WebAssembly');if(typeof caches==='undefined')why.push('sin almacenamiento en caché (¿contexto no seguro?)');if(typeof createImageBitmap==='undefined')why.push('sin createImageBitmap');
 const dm=typeof navigator!=='undefined'?navigator.deviceMemory:undefined;if(dm&&dm<PADDLE_LIMITS.minDeviceMemoryGB)why.push('memoria del dispositivo insuficiente ('+dm+' GB)');return {ok:why.length===0,why};}
async function loadManifest(){const r=await fetch(BASE+'MANIFEST.json',{cache:'no-cache'});if(!r.ok)throw Error('No se encuentra el lector avanzado en este servidor.');return r.json();}
const hex=buf=>[...new Uint8Array(buf)].map(b=>b.toString(16).padStart(2,'0')).join('');
/* Estado: instalado solo si el manifiesto guardado esta y TODOS sus ficheros estan en la cache con el tamano esperado. */
export async function paddleStatus(){try{const c=await caches.open(PADDLE_CACHE),m=await c.match(BASE+'MANIFEST.json');if(!m)return {installed:false};const man=await m.json();
  for(const f of man.files){for(const pth of [f.path,...(f.aliases||[])]){const hit0=await c.match(BASE+pth);if(!hit0)return {installed:false,partial:true};}const hit=await c.match(BASE+f.path);const len=Number(hit.headers.get('content-length')||-1);if(len>=0&&len!==f.bytes)return {installed:false,partial:true};}
  return {installed:true,version:man.version,totalBytes:man.totalBytes};}catch{return {installed:false};}}
/* Descarga con progreso; si algo falla o se cancela se deshace lo descargado (nunca queda un estado a medias que parezca instalado). */
export async function paddleInstall({onProgress,signal}={}){const sup=paddleSupport();if(!sup.ok)throw Error('Este dispositivo no admite el lector avanzado: '+sup.why.join(', ')+'.');
 const man=await loadManifest();await caches.delete(PADDLE_CACHE);/* limpio: el SW sirve la cache antes que la red, asi la descarga siempre viene del servidor */const c=await caches.open(PADDLE_CACHE);let done=0;const total=man.totalBytes,put=[];
 try{for(const f of man.files){if(signal&&signal.aborted)throw Error('Descarga cancelada.');const r=await fetch(BASE+f.path,{cache:'reload',signal});if(!r.ok)throw Error('No se pudo descargar '+f.path+' ('+r.status+').');
   const parts=[];let got=0;const rd=r.body.getReader();for(;;){const {done:d,value}=await rd.read();if(d)break;parts.push(value);got+=value.length;onProgress&&onProgress({done:done+got,total,file:f.path});}
   if(got!==f.bytes)throw Error('Descarga incompleta de '+f.path+'.');const blob=new Blob(parts,{type:ctype(f.path)});
   if(globalThis.crypto&&crypto.subtle){const h=hex(await crypto.subtle.digest('SHA-256',await blob.arrayBuffer()));if(h!==f.sha256)throw Error('La huella de '+f.path+' no coincide: descarga corrupta.');}
   for(const pth of [f.path,...(f.aliases||[])]){await c.put(BASE+pth,new Response(blob,{headers:{'content-type':ctype(f.path),'content-length':String(f.bytes)}}));put.push(pth);}done+=f.bytes;}
  await c.put(BASE+'MANIFEST.json',new Response(JSON.stringify(man),{headers:{'content-type':'application/json'}}));return {installed:true,totalBytes:total};}
 catch(e){for(const p of put)try{await c.delete(BASE+p);}catch{}throw e;}}
export async function paddleRemove(){try{return await caches.delete(PADDLE_CACHE);}catch{return false;}}
let live=null;
async function disposeLive(){const o=live;live=null;if(!o)return;try{await Promise.race([o.dispose(),new Promise(r=>setTimeout(r,PADDLE_LIMITS.disposeMs))]);}catch{}}
export const paddleIsLive=()=>live!==null;
export async function readTicketPaddle(file,{onProgress,signal,timeoutMs=PADDLE_LIMITS.timeoutMs}={}){
 const stopped=()=>{if(signal&&signal.aborted)throw Error('Lectura cancelada.');};stopped();const t0=performance.now(),m0=mem();
 if(!file||file.size>PADDLE_LIMITS.maxBytes)throw Error('El archivo es demasiado grande para leerlo en el teléfono (máximo 12 MB).');
 const bytes=new Uint8Array(await file.arrayBuffer()),type=sniffType(bytes);if(!['image/jpeg','image/png'].includes(type))throw Error('El lector avanzado lee fotos JPG o PNG.');
 const sup=paddleSupport();if(!sup.ok)throw Error('lector avanzado no disponible: '+sup.why.join(', '));
 let to=null,rejectStop;const stopP=new Promise((_,rej)=>{rejectStop=rej;});stopP.catch(()=>{});const guard=p=>Promise.race([p,stopP]);
 const onAbort=()=>rejectStop(Error('Lectura cancelada.'));if(signal)signal.addEventListener('abort',onAbort,{once:true});let timedOut=false;to=setTimeout(()=>{timedOut=true;rejectStop(Error('timeout'));},timeoutMs);
 try{onProgress&&onProgress({status:'Cargando el lector avanzado…'});await disposeLive();
  const {PaddleOCR}=await guard(import(BASE+'index.mjs'));
  const cp=PaddleOCR.create({worker:true,ortOptions:{backend:'wasm',wasmPaths:BASE+'ort/',numThreads:1,simd:true},textDetectionModelName:'PP-OCRv6_tiny_det',textDetectionModelAsset:{url:BASE+'models/PP-OCRv6_tiny_det_onnx_infer.tar'},textRecognitionModelName:'PP-OCRv6_tiny_rec',textRecognitionModelAsset:{url:BASE+'models/PP-OCRv6_tiny_rec_onnx_infer.tar'},textDetectionBatchSize:1,textRecognitionBatchSize:6});
  cp.then(o=>{if(signal&&signal.aborted||timedOut){try{o.dispose();}catch{}}},()=>{});live=await guard(cp);const t1=performance.now();
  onProgress&&onProgress({status:'Leyendo el ticket con el lector avanzado…'});stopped();
  const [res]=await guard(live.predict(new Blob([bytes],{type}),{textDetLimitType:'max',textDetLimitSideLen:PADDLE_LIMITS.sideLen}));const ocrMs=Math.round(performance.now()-t1);
  const items=(res&&res.items||[]).map(i=>({text:i.text,score:i.score,poly:i.poly})),lines=rowsFromBoxes(items);
  const meanConf=lines.length?Math.round(lines.reduce((a,l)=>a+l.conf,0)/lines.length):0,m1=mem();
  return {lines,passes:[lines],source:'ocr-paddle',engine:'paddle',pages:1,quality:{sharpness:null,blurry:false,meanConf,empty:lines.length===0},stats:{ms:Math.round(performance.now()-t0),ocrMs,heapMB:m1,heapDeltaMB:m0!==null&&m1!==null?m1-m0:null,boxes:items.length}};}
 catch(e){if(signal&&signal.aborted)throw Error('Lectura cancelada.');if(timedOut)throw Error('La lectura avanzada tarda demasiado en este teléfono.');throw e;}
 finally{clearTimeout(to);try{signal&&signal.removeEventListener('abort',onAbort);}catch{}await disposeLive();}}
